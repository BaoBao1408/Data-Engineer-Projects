create Kafka Topics: 
docker exec smartlog-kafka kafka-topics --create `
  --bootstrap-server localhost:9092 `
  --topic order.created `
  --partitions 3 `
  --replication-factor 1

docker exec smartlog-kafka kafka-topics --create `
  --bootstrap-server localhost:9092 `
  --topic order.status.updated `
  --partitions 3 `
  --replication-factor 1

docker exec -it smartlog-kafka kafka-consumer-groups `
  --bootstrap-server localhost:9092 `
  --group tracking-service-group `
  --reset-offsets `
  --to-earliest `
  --all-topics `
  --execute

Verify: docker exec smartlog-kafka kafka-topics --list --bootstrap-server localhost:9092
